-- SQL for Web Builder with Multiple Templates
-- Execute this in your Supabase SQL Editor

CREATE TABLE IF NOT EXISTS user_websites (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES directory_users(id) ON DELETE CASCADE UNIQUE,
    config JSONB DEFAULT '{}'::jsonb,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Note: The 'config' column stores the JSON object containing the selected template ID
-- and all its specific content (texts, images, list items, etc.).
